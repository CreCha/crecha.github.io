public sealed class AppointmentRepository : IAppointmentRepository
{
    private readonly IDbConnectionFactory _db;

    public AppointmentRepository(IDbConnectionFactory db) => _db = db;

    // Ελέγχει επικάλυψη ραντεβού πριν την καταχώριση
    public async Task<Result<int>> BookAsync(Appointment a, CancellationToken ct)
    {
        using var conn = await _db.OpenAsync(ct);
        using var tx = conn.BeginTransaction();

        var clash = await conn.ExecuteScalarAsync<bool>(@"
            SELECT CASE WHEN EXISTS (
              SELECT 1 FROM Appointments
              WHERE DoctorId = @DoctorId
                AND StartsAt < @EndsAt AND EndsAt > @StartsAt)
            THEN 1 ELSE 0 END", a, tx);

        if (clash) return Result<int>.Fail("Η ώρα δεν είναι διαθέσιμη");

        var id = await conn.InsertAsync(a, tx);
        tx.Commit();
        return Result<int>.Ok(id);
    }
}

WITH monthly AS (
    SELECT p.ProductName,
           DATEFROMPARTS(YEAR(o.OrderDate), MONTH(o.OrderDate), 1) AS Month,
           SUM(ol.Qty * ol.UnitPrice) AS Revenue
    FROM OrderLines ol
    JOIN Orders   o ON o.OrderId   = ol.OrderId
    JOIN Products p ON p.ProductId = ol.ProductId
    WHERE o.OrderDate >= DATEADD(YEAR, -1, GETDATE())
    GROUP BY p.ProductName, YEAR(o.OrderDate), MONTH(o.OrderDate)
)
SELECT ProductName, Month, Revenue,
       SUM(Revenue) OVER (PARTITION BY ProductName
                          ORDER BY Month) AS RunningTotal,
       RANK() OVER (PARTITION BY Month
                    ORDER BY Revenue DESC) AS MonthRank
FROM monthly
ORDER BY Month, MonthRank;
// Μικρός API client με timeout, retry και ελληνικά μηνύματα
export async function api(path, { retries = 2, ...opts } = {}) {
  for (let i = 0; i <= retries; i++) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), 8000);
    try {
      const res = await fetch(`/api/${path}`, {
        ...opts, signal: ctrl.signal,
        headers: { 'Content-Type': 'application/json' }
      });
      if (!res.ok) throw new Error(`Σφάλμα ${res.status}`);
      return await res.json();
    } catch (err) {
      if (i === retries) throw err;
      await new Promise(r => setTimeout(r, 500 * 2 ** i));
    } finally {
      clearTimeout(timer);
    }
  }
}
